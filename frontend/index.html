"""
manao.task — 提交云端任务到 K8S Job
   每个任务挂载用户代码 + 配额上下文,完成后写回 MongoDB / MinIO
"""
from kubernetes import client, config
from pydantic import BaseModel
from typing import Literal
from manao.core import ResourceQuota, QuotaExceeded


class TaskSpec(BaseModel):
    project_id: str
    runtime: Literal["python:3.11", "java:17", "node:20"]
    cpu_limit: str = "1000m"▎
    memory_limit: str = "2Gi"
    ttl_seconds: int = 3600


async def submit(spec: TaskSpec) -> str:
    """提交任务前先校验配额,再生成 Job 清单。"""
    quota = await ResourceQuota.fetch(spec.project_id)
    if not quota.has_capacity(spec):
        raise QuotaExceeded(f"项目 {spec.project_id} 已用尽配额")

    job = build_job_manifest(spec)
    api = client.BatchV1Api()
    created = api.create_namespaced_job(
        namespace=f"manao-{spec.project_id}",
        body=job,
    )
    return created.metadata.name
