import fs from 'node:fs';
import vm from 'node:vm';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';
import { isDeepStrictEqual } from 'node:util';
const dir = process.env.MANAO_DIAGNOSTIC_DIR;
if (!dir) throw new Error('MANAO_DIAGNOSTIC_DIR is required');
const port = Number(process.argv[2]);
const baseUrl = 'http://127.0.0.1:' + port;
const task = '00000000-0000-4000-8000-000000000001';
const fetched = spawnSync('kubectl', ['--request-timeout=15s','-n','manao-stage6b','exec','deploy/manao-pi-worker','--','cat','/app/dist/backend-client.js'], {maxBuffer:100000, timeout:20000});
if (fetched.status !== 0) throw new Error('worker module retrieval failed');
fs.writeFileSync(dir + '/deployed-backend-client.js', fetched.stdout);
const trace = [];
const context = vm.createContext({Buffer, AbortSignal, AbortController, URLSearchParams, setTimeout, clearTimeout, fetch});
const module = new vm.SourceTextModule(fetched.stdout.toString('utf8'), {context});
await module.link(specifier => {
  const values = specifier === 'node:util' ? {isDeepStrictEqual} : specifier === './live-output.js' ? {LIVE_OUTPUT_MAX_BYTES:262144} : {CONTEXT_PAGE_BYTES:4194304};
  return new vm.SyntheticModule(Object.keys(values), function(){ for(const [key,value] of Object.entries(values)) this.setExport(key,value); }, {context});
});
await module.evaluate();
const {BackendClient, ProgressChain} = module.namespace;
const client = new BackendClient({baseUrl, token:'diagnostic-only', instanceId:'diagnostic-worker',epoch:1,fetch:async(url,options)=>{
  const start=performance.now(); const response=await fetch(url,options); const body=await response.clone().json();
  const item=JSON.parse(options.body); trace.push({seq:item.seq,kind:item.kind,status:response.status,code:body.code??null,elapsedMs:Number((performance.now()-start).toFixed(2))}); return response;
}});
const chain = new ProgressChain(client, task);
for(let i=1;i<=36;i++)chain.send('tool_end','diagnostic existing event '+i);
await chain.drain();
const before=await fetch(baseUrl+'/state').then(r=>r.json());
const held=await fetch(baseUrl+'/recovery/start',{method:'POST'}).then(r=>r.json());
chain.send('message','the model has finished the task');
await chain.drain();
await chain.sendTerminal('COMPLETED','the model has finished the task',null);
const during=await fetch(baseUrl+'/state').then(r=>r.json());
await fetch(baseUrl+'/recovery/release',{method:'POST'});
const after=await fetch(baseUrl+'/state').then(r=>r.json());
await new Promise(r=>setTimeout(r,100));
const uncompensated=await fetch(baseUrl+'/state').then(r=>r.json());
const outcome={moduleSha256:createHash('sha256').update(fetched.stdout).digest('hex'),before,held,failedDeliveries:trace.filter(x=>x.seq>36),during,after,uncompensated,sendTerminalResolved:true,lastError:{name:chain.error()?.constructor.name,status:chain.error()?.status,code:chain.error()?.code}};
await client.progress(task,{seq:37,kind:'message',text:'the model has finished the task',truncated:false});
await client.progress(task,{seq:38,kind:'terminal',text:JSON.stringify({state:'COMPLETED',summary:'the model has finished the task',failureCode:null}),truncated:false});
outcome.identicalReplayAfterRelease=await fetch(baseUrl+'/state').then(r=>r.json());
fs.writeFileSync(dir+'/race-result.json',JSON.stringify(outcome,null,2),'utf8');
console.log(JSON.stringify(outcome));
